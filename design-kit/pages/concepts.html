groupby로 묶고 agg로 여러 값을 한 번에 구해요.

결과의 열 이름은 agg에 준 이름이 돼요. reset_index()로 펴 두면 다음 계산이 쉬워요.

concepts/pandas_basic.py12–15줄 그룹별 집계
import pandas as pd

# 데이터 읽기
df = pd.read_csv('cafe_sales.csv')
df.info()
print(df.head())

# 열 고르기와 조건
cold = df.loc[df['temp'] < 10, ['date', 'menu', 'qty']]
print(cold.shape)

# 그룹별 집계
summary = df.groupby('menu').agg(total=('qty', 'sum'),
                                 mean=('qty', 'mean'))
print(summary.reset_index())

# 피벗 테이블
pt = df.pivot_table(index='date', columns='weather', values='qty', aggfunc='sum')
print(pt.head())

# 결측치 세기
print(df.isna().sum())
# 결측치 채우기
df['qty'] = df['qty'].fillna(df.groupby('menu')['qty'].transform('median'))
